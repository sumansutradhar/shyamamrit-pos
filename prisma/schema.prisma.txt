generator client {
  provider = "prisma-client-js"
}

datasource db {
  provider = "mysql"
  url      = env("DATABASE_URL")
}

// ========== AUTH & USERS ==========
model User {
  id            Int           @id @default(autoincrement())
  name          String
  username      String        @unique
  password      String        // Hashed
  role          Role          @default(CASHIER)
  phone         String?
  isActive      Boolean       @default(true)
  shiftStart    DateTime?
  shiftEnd      DateTime?
  lastLogin     DateTime?
  createdAt     DateTime      @default(now())
  updatedAt     DateTime      @updatedAt

  // Relations
  orders        Order[]       @relation("CashierOrders")
  kotActions    KOTAction[]   @relation("StaffActions")
  expenses      Expense[]     @relation("ExpenseCreator")
  attendance    Attendance[]
  performance   StaffPerformance[]
  auditLogs     AuditLog[]    @relation("UserAudit")
  campaignWins  ScratchCardWin[] @relation("UserWins")

  @@map("users")
}

enum Role {
  ADMIN
  CASHIER
  WAITER
  KITCHEN
}

// ========== STAFF ATTENDANCE & PERFORMANCE ==========
model Attendance {
  id          Int       @id @default(autoincrement())
  userId      Int
  clockIn     DateTime
  clockOut    DateTime?
  totalHours  Float?    // Calculated in code
  date        DateTime  @default(now())
  user        User      @relation(fields: [userId], references: [id])

  @@map("attendances")
}

model StaffPerformance {
  id          Int       @id @default(autoincrement())
  userId      Int
  date        DateTime  @default(now())
  shiftHour   String    // e.g., "12PM-2PM"
  ordersHandled Int     @default(0)
  salesAmount  Float    @default(0)
  user        User      @relation(fields: [userId], references: [id])

  @@map("staff_performances")
}

// ========== TABLES & QR ==========
model Table {
  id          Int       @id @default(autoincrement())
  number      Int       @unique
  status      TableStatus @default(AVAILABLE)
  currentOrderId Int?   // FK to Order (if occupied)
  qrCode      String    @unique // Permanent QR hash
  createdAt   DateTime  @default(now())
  
  orders      Order[]   @relation("TableOrders")

  @@map("tables")
}

enum TableStatus {
  AVAILABLE
  OCCUPIED
  RESERVED
}

// ========== MENU & INVENTORY ==========
model Category {
  id          Int       @id @default(autoincrement())
  name        String    @unique
  displayOrder Int      @default(0)
  createdAt   DateTime  @default(now())
  
  items       MenuItem[]

  @@map("categories")
}

model MenuItem {
  id          Int       @id @default(autoincrement())
  name        String
  description String?
  price       Float
  categoryId  Int
  imageUrl    String?
  isAvailable Boolean   @default(true)
  isSatvik    Boolean   @default(false) // For Prasadam Filter
  spiceLevel  String?   // Mild, Medium, Spicy
  prepTime    Int?      // Minutes
  createdAt   DateTime  @default(now())
  updatedAt   DateTime  @updatedAt

  category    Category  @relation(fields: [categoryId], references: [id])
  orderItems  OrderItem[]
  ingredients Recipe[]

  @@map("menu_items")
}

// ========== INVENTORY & PURCHASE ==========
model Inventory {
  id          Int       @id @default(autoincrement())
  name        String
  category    String    // Dairy, Veg, Grain
  stock       Float     // Current stock in KG/Ltr
  unit        String    // KG, Ltr, Pcs
  minAlert    Float     @default(5)
  costPerUnit Float
  expiryDate  DateTime? // P1 Feature
  createdAt   DateTime  @default(now())
  updatedAt   DateTime  @updatedAt

  ingredients Recipe[]
  purchaseOrders PurchaseOrder[]

  @@map("inventories")
}

model Recipe {
  id          Int       @id @default(autoincrement())
  menuItemId  Int
  inventoryId Int
  quantity    Float     // Required quantity per dish
  menuItem    MenuItem  @relation(fields: [menuItemId], references: [id])
  inventory   Inventory @relation(fields: [inventoryId], references: [id])

  @@map("recipes")
}

model PurchaseOrder {
  id          Int       @id @default(autoincrement())
  inventoryId Int
  vendorName  String
  quantity    Float
  expectedDate DateTime
  status      String    @default("PENDING") // PENDING, DELIVERED
  createdAt   DateTime  @default(now())
  inventory   Inventory @relation(fields: [inventoryId], references: [id])

  @@map("purchase_orders")
}

// ========== ORDERS & KOT ==========
model Order {
  id              Int            @id @default(autoincrement())
  tableNumber     Int
  token           String         @unique // e.g., TBL-01-4567
  channel         OrderChannel   @default(DINEIN)
  customerName    String?
  customerPhone   String?
  subtotal        Float
  discountAmount  Float          @default(0)
  packagingFee    Float          @default(0)
  deliveryFee     Float          @default(0)
  total           Float
  cashPaid        Float?         // For partial payment
  onlinePaid      Float?         // For partial payment
  paymentMode     PaymentMode    @default(UNPAID)
  status          OrderStatus    @default(PENDING)
  createdAt       DateTime       @default(now())
  completedAt     DateTime?
  voidReason      String?        // P0 Mandatory on void

  cashierId       Int?
  table           Table          @relation(fields: [tableNumber], references: [number])
  cashier         User?          @relation("CashierOrders", fields: [cashierId], references: [id])
  items           OrderItem[]
  kotQueue        KOTQueue[]
  splitBills      BillSplit[]
  waitlist        Waitlist?      @relation("OrderWaitlist")
  broadcasts      OrderBroadcast?

  @@map("orders")
}

enum OrderChannel {
  DINEIN
  TAKEAWAY
  DELIVERY
}

enum PaymentMode {
  UNPAID
  CASH
  ONLINE
  HYBRID
}

enum OrderStatus {
  PENDING
  COOKING
  READY
  COMPLETED
  CANCELLED
}

model OrderItem {
  id          Int       @id @default(autoincrement())
  orderId     Int
  menuItemId  Int
  name        String    // Snapshot of item name
  price       Float     // Snapshot of price
  quantity    Int
  total       Float
  instructions String?  // "No Onion", "Extra Spicy" (P0)
  assignedTo  String?   // Person name for split billing (P0)

  order       Order     @relation(fields: [orderId], references: [id])
  menuItem    MenuItem  @relation(fields: [menuItemId], references: [id])

  @@map("order_items")
}

// ========== KITCHEN DISPLAY (KDS) ==========
model KOTQueue {
  id          Int       @id @default(autoincrement())
  orderId     Int
  tableNumber Int
  token       String
  itemsSummary String   // JSON string or text summary
  status      KOTStatus @default(PENDING)
  cookingTime Int?      // Seconds timer (P0)
  colorTag    String    // GREEN, ORANGE, BLUE (P0)
  createdAt   DateTime  @default(now())
  updatedAt   DateTime  @updatedAt

  order       Order     @relation(fields: [orderId], references: [id])

  @@map("kot_queue")
}

enum KOTStatus {
  PENDING
  COOKING
  READY
  CANCELLED
}

model KOTAction {
  id          Int       @id @default(autoincrement())
  kotId       Int
  staffId     Int
  action      String    // COOKING, READY, CANCELLED, VOID
  reason      String?   // Void Reason mandatory
  timestamp   DateTime  @default(now())
  staff       User      @relation("StaffActions", fields: [staffId], references: [id])
  kot         KOTQueue  @relation(fields: [kotId], references: [id])

  @@map("kot_actions")
}

// ========== BILL SPLIT (Itemized Group Split - P0) ==========
model BillSplit {
  id          Int       @id @default(autoincrement())
  orderId     Int
  personName  String
  assignedItems String   // JSON or comma separated item names
  subtotal    Float
  paid        Boolean   @default(false)
  order       Order     @relation(fields: [orderId], references: [id])

  @@map("bill_splits")
}

// ========== WAITLIST (P1) ==========
model Waitlist {
  id          Int       @id @default(autoincrement())
  customerName String
  customerPhone String
  guests      Int
  token       String    @unique
  status      String    @default("WAITING") // WAITING, SEATED, CANCELLED
  orderId     Int?      // When seated, this links to the order
  createdAt   DateTime  @default(now())
  notifiedAt  DateTime? // WhatsApp notification sent time
  order       Order?    @relation("OrderWaitlist", fields: [orderId], references: [id])

  @@map("waitlists")
}

// ========== CUSTOMER PROFILES (Taste Profile - P0) ==========
model CustomerProfile {
  id            Int       @id @default(autoincrement())
  phone         String    @unique
  name          String?
  noOnionGarlic Boolean   @default(false)
  spiceLevel    String    @default("MEDIUM") // MILD, MEDIUM, SPICY
  jainFood      Boolean   @default(false)
  preferences   String?   // JSON for extra customizations
  createdAt     DateTime  @default(now())
  updatedAt     DateTime  @updatedAt

  orders        Order[]   // Linked via phone match in logic
  scratchWins   ScratchCardWin[]

  @@map("customer_profiles")
}

  messageSent Boolean   @default(false)
  sentAt      DateTime?
  order       Order     @relation(fields: [orderId], references: [id])

  @@map("order_broadcasts")
}

  createdBy   Int

  itemName    String
  quantity    Float
  reason      String    // BURNT, RETURNED, EXPIRED
  cost        Float
  createdBy   Int

  @@map("wastage_logs")
}

  cashTotal   Float
  onlineTotal Float
  creditTotal Float
  expensesTotal Float
  netProfit   Float
  startFloat  Float     // P0 Cash Float
  endFloat    Float     // P0 Cash Float
  createdBy   Int
  createdAt   DateTime  @default(now())

  @@map("z_reports")
}

// ========== MARKETING (Scratch Card Campaigns) ==========
model Campaign {
  id              Int            @id @default(autoincrement())
  name            String
  startDate       DateTime
  endDate         DateTime
  minCartValue    Float          @default(500)
  dailyWinnerCap  Int            @default(5)
  isActive        Boolean        @default(true)
  createdAt       DateTime       @default(now())

  rewards         CampaignReward[]
  wins            ScratchCardWin[]

  @@map("campaigns")
}

model CampaignReward {
  id              Int       @id @default(autoincrement())
  campaignId      Int
  rewardType      String    // FREE_ITEM, PERCENTAGE, CASHBACK
  rewardValue     String    // e.g., "Gulab Jamun" or "10"
  quantity        Int       // Total pool quantity
  claimed         Int       @default(0)
  campaign        Campaign  @relation(fields: [campaignId], references: [id])

  @@map("campaign_rewards")
}

model ScratchCardWin {
  id              Int       @id @default(autoincrement())
  campaignId      Int
  customerPhone   String
  customerProfile CustomerProfile? @relation(fields: [customerPhone], references: [phone])
  rewardId        Int
  orderId         Int
  winDate         DateTime  @default(now())
  isClaimed       Boolean   @default(false)
  claimedAt       DateTime?
  campaign        Campaign  @relation(fields: [campaignId], references: [id])
  reward          CampaignReward @relation(fields: [rewardId], references: [id])

  @@map("scratch_card_wins")
}

// ========== SECURITY & AUDIT ==========
model AuditLog {
  id          Int       @id @default(autoincrement())
  userId      Int
  action      String    // LOGIN, ORDER_CREATE, VOID, PAYMENT, DELETE
  details     String?   // JSON
  ipAddress   String?
  timestamp   DateTime  @default(now())
  user        User      @relation("UserAudit", fields: [userId], references: [id])

  @@map("audit_logs")
}
